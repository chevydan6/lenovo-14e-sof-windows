cmake_minimum_required(VERSION 3.16)
project(chromebook_audio_dsp VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(chromebook_dsp SHARED src/chromebook_dsp.c)
set_target_properties(chromebook_dsp PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(chromebook_dsp PUBLIC include)
if(NOT MSVC)
  target_compile_options(chromebook_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(chromebook_dsp PRIVATE m)
endif()
add_executable(chromebook-dsp tools/process_wav.c)
target_link_libraries(chromebook-dsp PRIVATE chromebook_dsp)
if(NOT MSVC)
  target_compile_options(chromebook-dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(chromebook-dsp PRIVATE m)
endif()
option(CB_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(CB_SANITIZE AND NOT MSVC)
  foreach(target chromebook_dsp chromebook-dsp)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME dsp_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify.py $<TARGET_FILE:chromebook_dsp> $<TARGET_FILE:chromebook-dsp>)
endif()
