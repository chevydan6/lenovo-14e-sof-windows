cmake_minimum_required(VERSION 3.16)
project(lenovo14e_apo VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
option(LENOVO14E_APO_INTEGRITYCHECK "Require a signed deployment DLL" ON)
enable_language(RC)
add_library(lenovo14eapo SHARED apo_exports.def lenovo14e_apo.cpp lenovo14e_apo.rc ../audio-dsp/src/chromebook_dsp.c)
set_target_properties(lenovo14eapo PROPERTIES PREFIX "")
target_include_directories(lenovo14eapo PRIVATE ../audio-dsp/include)
target_compile_definitions(lenovo14eapo PRIVATE UNICODE _UNICODE)
target_link_libraries(lenovo14eapo PRIVATE ole32 uuid advapi32)
if(MSVC)
 target_link_options(lenovo14eapo PRIVATE /MANIFEST:NO)
 if(LENOVO14E_APO_INTEGRITYCHECK)
  target_link_options(lenovo14eapo PRIVATE /INTEGRITYCHECK)
 endif()
endif()
if(MINGW)
 target_compile_options(lenovo14eapo PRIVATE -Wall -Wextra -Wno-misleading-indentation)
 target_link_options(lenovo14eapo PRIVATE -static -static-libgcc -static-libstdc++)
 if(LENOVO14E_APO_INTEGRITYCHECK)
  target_link_options(lenovo14eapo PRIVATE -Wl,--forceinteg)
 endif()
endif()

add_executable(lenovo14e-apo-probe probe.cpp)
target_link_libraries(lenovo14e-apo-probe PRIVATE ole32 uuid advapi32)
if(MINGW)
 target_link_options(lenovo14e-apo-probe PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()

if(MSVC)
 set_property(TARGET lenovo14eapo lenovo14e-apo-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_executable(lenovo14e-apo-worker install_worker.cpp)
target_link_libraries(lenovo14e-apo-worker PRIVATE advapi32)
if(MINGW)
 target_link_options(lenovo14e-apo-worker PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
 set_property(TARGET lenovo14e-apo-worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
